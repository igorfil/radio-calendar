radio-calendar

A day view drawn as a vintage tuner dial. Drag the dial to peek at the rest of the day; click a meeting to fire event-click.

<script type="module">
  import "radio-calendar";
  const cal = document.querySelector("radio-calendar");
  cal.events = [
    { title: "Daily standup", start: "2026-11-13T09:15", end: "2026-11-13T09:45", category: "team" },
  ];
  cal.addEventListener("event-click", e => console.log(e.detail.event));
</script>
<radio-calendar theme="dark" work-start="09:00" work-end="17:00"></radio-calendar>
